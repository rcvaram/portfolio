Comparison operators 

<? php
$a = 2;
$b = 4;
//== checks whether the variable content/data is same or not;
if ($a == $b) {
  echo "They are equal";
}  // == basically check both content are same but in here this is false

if ($a!=$b){
  echo "They are not equal";
}
$b = "2";
if ($a == $b) {
  echo "They are equal";
}  // == basically check both  content are same It will be true

//=== checks whether varaible content/data + type of the variable is same

if ($a === $b) {
  echo "They are equal";
}  // == basically check both content & type are same, It will be false

if ($a !== $b) {
  echo "They are not equal";
}

// We have >, <, >=, <=, <> comparissions
?>

Conditional & Control Structures

  echo "a is less than b with bool is true";
}
else if (($a> $b) && $bool) {
  echo "a is greater than b with bool is true";
}
else {
  echo "Nothing meets";
}

//Switch expression strucure

  case 1:
    echo "a is one";
    break;
  case 2:
    echo "a is two";
    break;
  default:
    echo "This is a default behavour"
    break;
}

//Match expression structure

  1 => "variable a is equal to one";
  2,3,4 => "varaible is 2 or 3 or 5";
}; //This match means -> content + type or ->  ===
echo $result;

?>

Arrays 

<?php
//Way 1 to define array
$subjects = array("maths", "science", "tamil");

//Way 2 to define array
$subjects = ["maths", "science", "tamil"];

//To access the element use, indexes
echo $subjects[1] // "science"

//To add an element array - way 1
$subjects[] = "english"; // $subjects = ["maths", "science", "tamil", "english"];

//To add an element array - way 1
array_push()
//To replace an element in array
$subjects[2] = "drama"; //$subjects = ["maths", "science", "drama", "english"];

//To remmove
unset($subjects[2]); // No re-ordering in array as other language, so, index 2 will be deleted.

//To reordering when we remove
array_splice($subjects, 0, 1);  

// count elements
count($subjects); //4

// sort elements
sort($subjects);


//Array Keys / Lables

$tasks = ["siva" => "work", "ram" => "study"];
//Here indexes will be replaced as keys
echo $tasks["siva"]; // "work"

print_r($tasks); // Array([siva]=> work [ram]=> study)

//array_push will not work in keys array, we need to use tasks["new_val"]
 
?>
